// ===========================================
// GET AIRPORT CODE
// ===========================================
function getAirportCode(city) {
    const lowerCity = city.toLowerCase().trim();
    
    // Direct match
    if (flightDatabase.airports[lowerCity]) {
        return flightDatabase.airports[lowerCity];
    }
    
    // Partial match
    for (let [key, code] of Object.entries(flightDatabase.airports)) {
        if (lowerCity.includes(key) || key.includes(lowerCity)) {
            return code;
        }
    }
    
    console.log(`⚠️ No airport code for: ${city}, using default JFK`);
    return "JFK";
}

// ===========================================
// GET FLIGHT INFO - MAIN FUNCTION
// ===========================================
function getFlightInfo(origin, destination) {
    console.log(`🔍 Getting flight info: ${origin} → ${destination}`);
    
    const originCode = getAirportCode(origin);
    const destCode = getAirportCode(destination);
    
    console.log(`✈️ Airport codes: ${originCode} → ${destCode}`);
    
    const routeKey = `${originCode}-${destCode}`;
    const reverseKey = `${destCode}-${originCode}`;
    
    // Look up route
    let routeData = flightDatabase.routes[routeKey];
    
    // Try reverse route
    if (!routeData) {
        routeData = flightDatabase.routes[reverseKey];
        if (routeData) {
            console.log(`🔄 Using reverse route: ${reverseKey}`);
        }
    }
    
    // Return data or default
    if (routeData) {
        return {
            price: routeData.price,
            airline: routeData.airlines,
            originCode: originCode,
            destCode: destCode
        };
    }
    
    // Default fallback based on distance
    let defaultPrice = 500;
    const longHaulDests = ["NRT", "SYD", "LAX", "SFO", "JFK"];
    const mediumHaulDests = ["DXB", "BKK", "SIN", "DPS"];
    
    if (longHaulDests.includes(destCode) || longHaulDests.includes(originCode)) {
        defaultPrice = 1000;
    } else if (mediumHaulDests.includes(destCode) || mediumHaulDests.includes(originCode)) {
        defaultPrice = 600;
    }
    
    return {
        price: defaultPrice,
        airline: "Multiple airlines available",
        originCode: originCode,
        destCode: destCode
    };
}
